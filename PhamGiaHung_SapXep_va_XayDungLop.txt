#pragma once
#include <cstdio>
#include <ctime>
#include <string>
using namespace std;

struct Date {
    int d = 1, m = 1, y = 2000;

    long key() const { return y * 10000L + m * 100 + d; }
    string str() const {
        char buf[16];
        snprintf(buf, sizeof(buf), "%02d/%02d/%04d", d, m, y);
        return buf;
    }
    static Date homNay() {
        time_t t = time(nullptr);
        tm* lt = localtime(&t);
        Date r;
        r.d = lt->tm_mday;
        r.m = lt->tm_mon + 1;
        r.y = lt->tm_year + 1900;
        return r;
    }
    static bool hopLe(int d, int m, int y) {
        if (y < 1900 || y > 2200 || m < 1 || m > 12 || d < 1) return false;
        int ngay[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
        bool nhuan = (y % 4 == 0 && y % 100 != 0) || (y % 400 == 0);
        int toiDa = ngay[m - 1] + ((m == 2 && nhuan) ? 1 : 0);
        return d <= toiDa;
    }
};

// ============================================================================
// QuickSort.h - Thuat toan Quick Sort tu cai dat (template)
// Nguoi phu trach: P  (phan "Sap xep du lieu")
// ============================================================================
#pragma once
#include <algorithm>
#include <vector>
using namespace std;

// Quick Sort khong on dinh, nen ham so sanh luon co them tieu chi phu (ma mat hang).
template <typename T, typename Truoc>
static void quickSort(vector<T>& a, int lo, int hi, Truoc truoc) {
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (truoc(a[mid], a[lo])) swap(a[mid], a[lo]);
        if (truoc(a[hi], a[lo])) swap(a[hi], a[lo]);
        if (truoc(a[hi], a[mid])) swap(a[hi], a[mid]);
        T pivot = a[mid];
        int i = lo, j = hi;
        while (i <= j) {
            while (truoc(a[i], pivot)) i++;
            while (truoc(pivot, a[j])) j--;
            if (i <= j) { swap(a[i], a[j]); i++; j--; }
        }
        if (j - lo < hi - i) { quickSort(a, lo, j, truoc); lo = i; }
        else { quickSort(a, i, hi, truoc); hi = j; }
    }
}

#pragma once

// ---- TonKho: mot lo hang cua mat hang ----
class TonKho {
private:
    string maLoHang;
    int soLuongTon;
    Date hanSuDung;
    string viTriKho;

public:
    TonKho() : soLuongTon(0) {}
    TonKho(const string& ma, int sl, const Date& hsd, const string& viTri)
        : maLoHang(ma), soLuongTon(sl), hanSuDung(hsd), viTriKho(viTri) {}

    // ---- getter ----
    const string& getMaLoHang() const { return maLoHang; }
    int getSoLuongTon() const { return soLuongTon; }
    const Date& getHanSuDung() const { return hanSuDung; }
    const string& getViTriKho() const { return viTriKho; }

    // ---- setter ----
    void setHanSuDung(const Date& d) { hanSuDung = d; }
    void setViTriKho(const string& v) { viTriKho = v; }

    // true neu lo hang con han su dung
    bool kiemTraHanSD() const { return hanSuDung.key() >= Date::homNay().key(); }
    // qty > 0: nhap them, qty < 0: xuat bot
    bool capNhatSoLuong(int qty) {
        if (soLuongTon + qty < 0) return false;
        soLuongTon += qty;
        return true;
    }
};

// ---- MatHang ----
class MatHang {
private:
    string maSanPham;
    string tenSanPham;
    double giaBan;
    int mucTonToiThieu;
    vector<TonKho> cacLo;  // 1 mat hang - nhieu lo ton kho

public:
    MatHang() : giaBan(0), mucTonToiThieu(0) {}
    MatHang(const string& ma, const string& ten, double gia, int mucMin)
        : maSanPham(ma), tenSanPham(ten), giaBan(gia), mucTonToiThieu(mucMin) {}

    // ---- getter ----
    const string& getMaSanPham() const { return maSanPham; }
    const string& getTenSanPham() const { return tenSanPham; }
    double getGiaBan() const { return giaBan; }
    int getMucTonToiThieu() const { return mucTonToiThieu; }
    const vector<TonKho>& getCacLo() const { return cacLo; }

    // ---- setter ----
    void setTenSanPham(const string& ten) { tenSanPham = ten; }
    void setMucTonToiThieu(int v) { mucTonToiThieu = v; }
    void capNhatGia(double giaMoi) { giaBan = giaMoi; }

    // ---- nhap / xuat (ma da duoc dat truoc, co kiem tra du lieu nhap) ----
    void nhap() {
        tenSanPham = docChuoi("Ten mat hang: ");
        giaBan = docThuc("Gia ban (d): ", 0);
        mucTonToiThieu = docSo("Muc ton toi thieu: ", 0, 1000000);
    }
    static void xuatTieuDe() {
        cout << left << setw(8) << "Ma" << setw(26) << "Ten mat hang"
             << right << setw(12) << "Gia ban" << setw(8) << "Ton"
             << setw(8) << "Min" << "  TT\n";
        cout << string(70, '-') << "\n";
    }
    void xuat() const {
        cout << left << setw(8) << maSanPham << setw(26) << tenSanPham.substr(0, 25)
             << right << setw(12) << (long long)giaBan << setw(8) << tongTon()
             << setw(8) << mucTonToiThieu << "  "
             << (kiemTraCanCanhBao() ? "SAP HET!" : "OK") << "\n";
    }
    void xuatChiTietLo() const {
        if (cacLo.empty()) { cout << "    (chua co lo hang)\n"; return; }
        for (const auto& l : cacLo) {
            cout << "    - Lo " << l.getMaLoHang() << ": SL=" << l.getSoLuongTon()
                 << ", HSD=" << l.getHanSuDung().str() << ", vi tri=" << l.getViTriKho()
                 << (l.kiemTraHanSD() ? "" : "  [HET HAN]") << "\n";
        }
    }

    // ---- nghiep vu ton kho ----
    int tongTon() const {
        int t = 0;
        for (const auto& l : cacLo) t += l.getSoLuongTon();
        return t;
    }
    // so luong co the xuat (chi tinh lo con han)
    int tonXuatDuoc() const {
        int t = 0;
        for (const auto& l : cacLo)
            if (l.kiemTraHanSD()) t += l.getSoLuongTon();
        return t;
    }
    // gia tri ton kho cua mat hang = tong ton * gia ban
    double tinhGiaTriTon() const { return tongTon() * giaBan; }
    // gia tri cua cac lo da het han
    double giaTriHetHan() const {
        double t = 0;
        for (const auto& l : cacLo)
            if (!l.kiemTraHanSD()) t += l.getSoLuongTon() * giaBan;
        return t;
    }
    bool kiemTraCanCanhBao() const { return tongTon() <= mucTonToiThieu; }

    // Xuat kho theo FEFO: lo het han som nhat xuat truoc, bo qua lo qua han
    bool xuatKho(int soLuong) {
        if (soLuong <= 0 || soLuong > tonXuatDuoc()) return false;
        vector<TonKho*> ds;
        for (auto& l : cacLo)
            if (l.kiemTraHanSD() && l.getSoLuongTon() > 0) ds.push_back(&l);
        sort(ds.begin(), ds.end(), [](TonKho* a, TonKho* b) {
            return a->getHanSuDung().key() < b->getHanSuDung().key();
        });
        for (TonKho* l : ds) {
            if (soLuong == 0) break;
            int lay = min(soLuong, l->getSoLuongTon());
            l->capNhatSoLuong(-lay);
            soLuong -= lay;
        }
        return true;
    }
    // Nhap kho: chi cong don khi trung ma lo VA cung han su dung (cung 1 lo that su).
    // Trung ma lo nhung KHAC han su dung -> tu choi (tra ve false), khong ghi de han cu.
    bool nhapKho(const TonKho& lo) {
        for (auto& l : cacLo) {
            if (l.getMaLoHang() == lo.getMaLoHang()) {
                if (l.getHanSuDung().key() != lo.getHanSuDung().key()) return false;
                l.capNhatSoLuong(lo.getSoLuongTon());
                l.setViTriKho(lo.getViTriKho());
                return true;
            }
        }
        cacLo.push_back(lo);
        return true;
    }
    // Huy cac lo da het han: xoa khoi mat hang, dua cac lo con hang vao 'daHuy'
    // (de lap phieu xuat huy). Tra ve tong so luong da huy.
    int huyLoHetHan(vector<TonKho>& daHuy) {
        int huy = 0;
        for (size_t i = 0; i < cacLo.size();) {
            if (!cacLo[i].kiemTraHanSD()) {
                if (cacLo[i].getSoLuongTon() > 0) daHuy.push_back(cacLo[i]);
                huy += cacLo[i].getSoLuongTon();
                cacLo.erase(cacLo.begin() + i);
            } else {
                i++;
            }
        }
        return huy;
    }
};

// ============================================================================
// SapXep.h - Sap xep du lieu bang Quick Sort:
//   - Sap xep theo gia ban (tang dan / giam dan)
//   - Sap xep theo so luong ton kho (tang dan / giam dan)
// (Cai dat cac ham thanh vien cua lop QuanLyKho)
// Nguoi phu trach: P
// ============================================================================
#pragma once
#include <iostream>
#include <vector>
#include "QuanLyKho.h"
#include "QuickSort.h"
using namespace std;

// Lay danh sach mat hang (LNR) roi sap xep bang QUICK SORT.
// Khi gia tri bang nhau thi xep theo ma mat hang tang dan (ket qua luon on dinh, de kiem tra).
inline vector<MatHang*> QuanLyKho::sapXepTheoGia(bool tang) {
    vector<MatHang*> ds = cay.duyetTheoThuTu();
    quickSort(ds, 0, (int)ds.size() - 1, [tang](MatHang* a, MatHang* b) {
        if (a->getGiaBan() != b->getGiaBan())
            return tang ? a->getGiaBan() < b->getGiaBan() : a->getGiaBan() > b->getGiaBan();
        return a->getMaSanPham() < b->getMaSanPham();
    });
    return ds;
}

inline vector<MatHang*> QuanLyKho::sapXepTheoTon(bool tang) {
    vector<MatHang*> ds = cay.duyetTheoThuTu();
    quickSort(ds, 0, (int)ds.size() - 1, [tang](MatHang* a, MatHang* b) {
        if (a->tongTon() != b->tongTon())
            return tang ? a->tongTon() < b->tongTon() : a->tongTon() > b->tongTon();
        return a->getMaSanPham() < b->getMaSanPham();
    });
    return ds;
}

inline void QuanLyKho::sapXepDuLieu() {
    cout << "\n--- SAP XEP DU LIEU (thuat toan Quick Sort) ---\n";
    if (cay.rong()) { cout << "  Kho dang rong.\n"; return; }
    cout << "  1. Theo gia ban - tang dan\n"
         << "  2. Theo gia ban - giam dan\n"
         << "  3. Theo so luong ton kho - tang dan\n"
         << "  4. Theo so luong ton kho - giam dan\n"
         << "  0. Quay lai\n";
    int c = docSo("Chon: ", 0, 4);
    if (c == 0) return;
    vector<MatHang*> ds;
    switch (c) {
        case 1: cout << "\nSap xep theo GIA BAN tang dan:\n"; ds = sapXepTheoGia(true); break;
        case 2: cout << "\nSap xep theo GIA BAN giam dan:\n"; ds = sapXepTheoGia(false); break;
        case 3: cout << "\nSap xep theo TON KHO tang dan:\n"; ds = sapXepTheoTon(true); break;
        default: cout << "\nSap xep theo TON KHO giam dan:\n"; ds = sapXepTheoTon(false); break;
    }
    inDanhSach(ds);
}
